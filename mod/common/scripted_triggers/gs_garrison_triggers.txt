# State scope. No field armies are counted.
gs_state_controlled = {
    exists = owner
    exists = controller
    controller = owner
    state_population > 0
}

gs_state_service_eligible = {
    gs_state_controlled = yes
    any_scope_building = {
        is_building_type = building_gs_garrison
        OR = {
            has_active_production_method = pm_gs_active
            has_active_production_method = pm_gs_civic
        }
        building_has_goods_shortage = no
    }
}

gs_security_qualified = {
    var:gs_coverage_snapshot >= gs_deradicalization_minimum_coverage
}

gs_civic_qualified = {
    var:gs_coverage_snapshot >= gs_loyalty_minimum_coverage
    any_scope_building = {
        is_building_type = building_gs_garrison
        has_active_production_method = pm_gs_civic
        building_has_goods_shortage = no
    }
    is_incorporated = yes
    var:gs_radical_share <= gs_loyalty_max_turmoil
    population_below_expected_sol <= gs_maximum_population_below_expected_sol
}
